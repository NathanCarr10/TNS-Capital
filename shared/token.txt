# Log in and keep both tokens
RESPONSE=$(curl -s -X POST http://localhost:4000/auth/login \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"adminPassword"}')
TOKEN=$(echo "$RESPONSE" | grep -o '"accessToken":"[^"]*"' | cut -d'"' -f4)
REFRESH=$(echo "$RESPONSE" | grep -o '"refreshToken":"[^"]*"' | cut -d'"' -f4)

echo $TOKEN

curl -H "Authorization: Bearer $TOKEN" http://localhost:4000/auth/me
curl -H "Authorization: Bearer $TOKEN" http://localhost:3000/your-protected-endpoint

# New access token once the old one expires; then log out
curl -X POST localhost:4000/auth/refresh -H "Content-Type: application/json" -d "{\"refreshToken\":\"$REFRESH\"}"
curl -X POST localhost:4000/auth/logout -H "Content-Type: application/json" -d "{\"refreshToken\":\"$REFRESH\"}"

curl -X POST localhost:4000/auth/register -H "Content-Type: application/json" -d '{"username":"trader1","password":"password123"}'
